#07 实战：新增自定义 QOM 类型

本讲将所有前序知识串联成一个 完整可编译的自定义 QOM 类型。 我们会实现一个简单的 "hello-device"：一个 SysBus 设备，有两个属性（message 和 repeat）， 每次 MMIO 读取返回不同内容。

7.1 Header: 声明类型

/* include/hw/misc/hello.h */
#ifndef HELLO_H
#define HELLO_H

#include "hw/sysbus.h"
#include "qom/object.h"

#define TYPE_HELLO_DEVICE "hello-device"

/* 步骤 1: 用 OBJECT_DECLARE_TYPE 声明类型, 自动生成 cast helpers */
OBJECT_DECLARE_TYPE(HelloDeviceState, HelloDeviceClass, HELLO_DEVICE)

/* 步骤 2: 定义 Instance struct —— 首字段必须是父类实例 */
struct HelloDeviceState {
    SysBusDevice parent_obj;   /* ★ 必须是第一个字段 */

    /* 我们的自定义字段 */
    MemoryRegion mmio;
    char *message;             /* 属性: "message" */
    uint32_t repeat;           /* 属性: "repeat" */
    uint32_t count;            /* 内部状态: MMIO 读取次数 */
};

/* 步骤 3: 定义 Class struct —— 首字段必须是父类 class */
struct HelloDeviceClass {
    SysBusDeviceClass parent_class;  /* ★ 必须是第一个字段 */

    /* 如果有虚函数, 在这里声明。简单设备通常不需要 */
};

#endif /* HELLO_H */
OBJECT_DECLARE_TYPE(HelloDeviceState, HelloDeviceClass, HELLO_DEVICE) 生成了什么？
  • HELLO_DEVICE(obj) → 安全的 Object* → HelloDeviceState* 转换
  • HELLO_DEVICE_GET_CLASS(obj) → 获取 HelloDeviceClass
  • HELLO_DEVICE_CLASS(klass) → 安全的 ObjectClass* → HelloDeviceClass* 转换
  • g_autoptr(HelloDeviceState) → 自动引用计数释放

7.2 Source: 实现类型

/* hw/misc/hello.c */
#include "qemu/osdep.h"
#include "hw/misc/hello.h"
#include "qemu/log.h"

/* ── MMIO 回调 ── */
static uint64_t hello_mmio_read(void *opaque, hwaddr addr, unsigned size)
{
    HelloDeviceState *s = HELLO_DEVICE(opaque);
    const char *msg = s->message ? s->message : "Hello from QOM!";

    s->count++;
    return (uint64_t)msg[(s->count / s->repeat) % strlen(msg)];
}

static const MemoryRegionOps hello_mmio_ops = {
    .read = hello_mmio_read,
    .endianness = DEVICE_NATIVE_ENDIAN,
};

/* ── 属性 getter/setter ── */
static void hello_get_message(Object *obj, Visitor *v, const char *name,
                                void *opaque, Error **errp)
{
    HelloDeviceState *s = HELLO_DEVICE(obj);
    visit_type_str(v, name, &s->message, errp);
}

static void hello_set_message(Object *obj, Visitor *v, const char *name,
                                void *opaque, Error **errp)
{
    HelloDeviceState *s = HELLO_DEVICE(obj);
    char *value = NULL;

    if (!visit_type_str(v, name, &value, errp)) {
        return;
    }
    g_free(s->message);
    s->message = value;
}

static void hello_get_repeat(Object *obj, Visitor *v, const char *name,
                               void *opaque, Error **errp)
{
    HelloDeviceState *s = HELLO_DEVICE(obj);
    visit_type_uint32(v, name, &s->repeat, errp);
}

static void hello_set_repeat(Object *obj, Visitor *v, const char *name,
                               void *opaque, Error **errp)
{
    HelloDeviceState *s = HELLO_DEVICE(obj);
    uint32_t value;
    if (!visit_type_uint32(v, name, &value, errp)) {
        return;
    }
    if (value == 0) {
        error_setg(errp, "repeat must be > 0");  /* setter 可以实现验证 */
        return;
    }
    s->repeat = value;
}

/* ── instance_init: 不可失败, 只做简单的初始化 ── */
static void hello_device_init(Object *obj)
{
    HelloDeviceState *s = HELLO_DEVICE(obj);

    s->message = g_strdup("Hello from QOM!");
    s->repeat = 1;
    s->count = 0;

    /* 注册属性 */
    object_property_add(obj, "message", "str",
                        hello_get_message, hello_set_message, NULL, s);
    object_property_add(obj, "repeat", "uint32",
                        hello_get_repeat, hello_set_repeat, NULL, s);
}

/* ── realize: 可失败, 分配硬件资源 ── */
static void hello_device_realize(DeviceState *dev, Error **errp)
{
    HelloDeviceState *s = HELLO_DEVICE(dev);

    /* 注册 MMIO 区域 —— 这是真的硬件资源, 可以失败 */
    memory_region_init_io(&s->mmio, OBJECT(dev), &hello_mmio_ops, s,
                          "hello-mmio", 0x1000);
    sysbus_init_mmio(SYS_BUS_DEVICE(dev), &s->mmio);
}

/* ── finalize: 释放资源 ── */
static void hello_device_finalize(Object *obj)
{
    HelloDeviceState *s = HELLO_DEVICE(obj);
    g_free(s->message);
}

/* ── class_init: 设置虚函数指针 ── */
static void hello_device_class_init(ObjectClass *oc, void *data)
{
    DeviceClass *dc = DEVICE_CLASS(oc);
    dc->realize = hello_device_realize;   /* 覆盖 realize 虚函数 */
    dc->desc = "Hello QOM Device";
    dc->user_creatable = true;           /* 允许 device_add */
}

/* ── ★ 核心: 这个宏生成 TypeInfo + 注册函数 + type_init ── */
OBJECT_DEFINE_TYPE(HelloDeviceState, hello_device, HELLO_DEVICE, SYS_BUS_DEVICE)

7.3 OBJECT_DEFINE_TYPE 展开全景

OBJECT_DEFINE_TYPE(HelloDeviceState, hello_device, HELLO_DEVICE, SYS_BUS_DEVICE) 展开后的效果： 

/* 等同于: OBJECT_DEFINE_TYPE_EXTENDED(HelloDeviceState, hello_device,
                                       HELLO_DEVICE, SYS_BUS_DEVICE,
                                       false, sizeof(HelloDeviceClass), ) */

static void hello_device_finalize(Object *obj);              /* 声明 */
static void hello_device_class_init(ObjectClass *oc, const void *data);
static void hello_device_init(Object *obj);

static const TypeInfo hello_device_info = {
    .parent           = TYPE_SYS_BUS_DEVICE,       /* ← 继承自 SysBusDevice */
    .name             = "hello-device",
    .instance_size    = sizeof(HelloDeviceState),
    .instance_align   = __alignof__(HelloDeviceState),
    .instance_init    = hello_device_init,
    .instance_finalize = hello_device_finalize,
    .class_size       = sizeof(HelloDeviceClass),
    .class_init       = hello_device_class_init,
    .abstract         = false,                   /* 可实例化 */
};

static void hello_device_register_types(void)
{
    type_register_static(&hello_device_info);
}
type_init(hello_device_register_types);             /* ★ 构造时注册 */
关键点: 开发者只需提供三个函数 (_init, _class_init, _finalize) 和一个 struct (HelloDeviceState)。其余全部由宏自动生成。这就是 QOM 的编程范式。

7.4 构建集成

将新设备加入 QEMU 构建系统 (meson.build):

/* hw/misc/meson.build */
system_ss.add(when: 'CONFIG_HELLO_DEVICE', if_true: files('hello.c'))

/* 或者简单加 (无条件编译): */
system_ss.add(files('hello.c'))

7.5 运行时交互

启动 QEMU 并验证我们的设备：

# 启动 QEMU, 添加 hello-device
$ qemu-system-x86_64 -M pc -device hello-device,id=h1,message="QOM rocks!",repeat=3

# 在 QMP monitor 中自省
(qemu) info qom-tree
/machine/peripheral/h1 (hello-device)
  /message = "QOM rocks!"
  /repeat = 3

(qemu) qom-get /machine/peripheral/h1 message
"QOM rocks!"

(qemu) qom-set /machine/peripheral/h1 message "Learning QOM"
(qemu) qom-get /machine/peripheral/h1 message
"Learning QOM"

(qemu) object_add hello-device,id=h2,message="Dynamic!",repeat=1
# 第二个设备被动态创建！

7.7 调试与排查

常见问题与定位方法：

症状原因排查
"unknown type 'hello-device'"类型未注册检查 type_init 是否正确展开，module 是否加载
SIGSEGV in instance_initinstance struct 首字段不是 parent确认第一个字段是 SysBusDevice parent_obj
属性设置失败属性类型不匹配用 qom-list 检查属性类型字符串
realize 未被调用设备抽象 (abstract=true)检查 OBJECT_DEFINE_TYPE 中 abstract 参数
device_add 不工作user_creatable 未设置在 class_init 中设置 dc->user_creatable = true

本模块小结

步骤做什么用到的宏/函数
1. Header声明类型名、Instance/Class structsOBJECT_DECLARE_TYPE
2. instance_init初始化字段默认值、注册属性object_property_add
3. class_init设置虚函数指针、static 属性dc->realize = ...
4. realize分配硬件资源 (MMIO, IRQ, ...)memory_region_init_io
5. 注册宏生成 TypeInfo + type_initOBJECT_DEFINE_TYPE
6. 构建加入编译系统meson.build
7. 验证运行时自省info qom-tree, qom-get/set
