#05 继承、接口与类型转换

5.1 Type Hierarchy —— 继承链

QOM 的继承通过 TypeInfo 的 .parent 字段声明。看看 QEMU 的核心继承链：

TYPE_OBJECT ("object") ← 根类型, abstract │ ├─ TYPE_DEVICE ("device") ← 所有设备的基类, abstract │ │ │ ├─ TYPE_SYS_BUS_DEVICE ← 挂 system bus 的设备, abstract │ │ ├─ "edu" ← 一个具体设备 │ │ ├─ "serial" ← 16550 UART │ │ └─ ... │ │ │ ├─ TYPE_PCI_DEVICE ← PCI 设备基类, abstract │ │ ├─ "e1000" ← Intel e1000 网卡 │ │ ├─ "virtio-net-pci" ← virtio-net PCI 适配器 │ │ └─ ... │ │ │ └─ (更多设备子类型...) │ ├─ TYPE_BUS ("bus") ← 总线基类 │ ├─ TYPE_PCI_BUS ("PCI") ← PCI 总线 │ ├─ TYPE_SYSTEM_BUS ← 系统总线 │ └─ TYPE_I2C_BUS ← I2C 总线 │ ├─ TYPE_MACHINE ("machine") ← Machine 类型 │ ├─ "virt-machine" ← ARM virt │ ├─ "pc-q35-*" ← x86 Q35 │ └─ ... │ ├─ TYPE_MEMORY_BACKEND ("memory-backend") │ ├─ "memory-backend-ram" │ ├─ "memory-backend-file" │ └─ "memory-backend-memfd" │ └─ TYPE_CONTAINER ("container")

5.2 继承如何在 Class 层面实现

Class 的继承通过 memcpy + 覆盖 实现（参见 #03 的 type_initialize 流程）。 核心技术约束：

  1. Instance 首字段必须是 parent instance struct —— 保证向上转型安全
  2. Class 首字段必须是 parent class struct —— 保证父类函数指针被正确继承
  3. 子类 class_size ≥ 父类 class_size —— memcpy 的基础
/* DeviceState 的首字段是 Object (通过包含 parent_obj) */
struct DeviceState {
    Object parent_obj;   /* ★ 必须是第一个字段！ */
    const char *id;
    bool realized;
    ...
};

/* DeviceClass 的首字段是 ObjectClass */
struct DeviceClass {
    ObjectClass parent_class;   /* ★ 必须是第一个字段！ */
    DeviceRealize realize;
    DeviceUnrealize unrealize;
    ...
};
为什么这种结构保证安全？ &dev->parent_obj == (Object *)dev 因为 parent_obj 在偏移 0 处。 同样 &dc->parent_class == (ObjectClass *)dc。 这就是 C 语言中实现 static_cast<Base*> 的根本条件。

5.3 OBJECT_DECLARE_TYPE 宏 —— 类型安全转换

手动写 (DeviceState *)obj 是不安全的。QOM 提供了一套 C 宏生成类型安全的转换函数：

/* include/qom/object.h:234 */
#define OBJECT_DECLARE_TYPE(InstanceType, ClassType, MODULE_OBJ_NAME)
    typedef struct InstanceType InstanceType;
    typedef struct ClassType ClassType;
    G_DEFINE_AUTOPTR_CLEANUP_FUNC(InstanceType, object_unref)
    DECLARE_OBJ_CHECKERS(InstanceType, ClassType, MODULE_OBJ_NAME, \
                         TYPE_##MODULE_OBJ_NAME)

以 OBJECT_DECLARE_TYPE(DeviceState, DeviceClass, DEVICE) 为例，展开后生成：

/* 已展开: OBJECT_DECLARE_TYPE(DeviceState, DeviceClass, DEVICE) */

/* 1. typedefs */
typedef struct DeviceState DeviceState;
typedef struct DeviceClass DeviceClass;

/* 2. g_autoptr 支持 —— 离开作用域自动 object_unref */
G_DEFINE_AUTOPTR_CLEANUP_FUNC(DeviceState, object_unref)

/* 3. 类型安全转换函数 */

/* Instance checker: Object* → DeviceState* */
static inline DeviceState *DEVICE(const void *obj) {
    return OBJECT_CHECK(DeviceState, obj, TYPE_DEVICE);
}

/* Class getter: Object* → DeviceClass* */
static inline DeviceClass *DEVICE_GET_CLASS(const void *obj) {
    return OBJECT_GET_CLASS(DeviceClass, obj, TYPE_DEVICE);
}

/* Class checker: ObjectClass* → DeviceClass* */
static inline DeviceClass *DEVICE_CLASS(const void *klass) {
    return OBJECT_CLASS_CHECK(DeviceClass, klass, TYPE_DEVICE);
}

使用时：

void some_func(Object *obj)
{
    DeviceState *dev = DEVICE(obj);       /* 运行时检查: obj 确实是 TYPE_DEVICE 的子类型? */
    DeviceClass *dc = DEVICE_GET_CLASS(obj); /* 通过 obj->class 获取 DeviceClass */
}

5.4 OBJECT_CHECK —— 运行时类型检查

/* include/qom/object.h:531 */
#define OBJECT_CHECK(type, obj, name)
    ((type *)object_dynamic_cast_assert(OBJECT(obj), (name), \
                                        __FILE__, __LINE__, __func__))

object_dynamic_cast 是运行时类型检查的核心：

static Object *object_dynamic_cast_assert(Object *obj, const char *typename,
                                          const char *file, int line,
                                          const char *func)
{
    Object *inst;
    for (int i = 0; obj && i < OBJECT_CLASS_CAST_CACHE; i++) {
        if (obj->class->object_cast_cache[i] == typename) {
            goto out;  /* cache hit —— 绝大多数情况 */
        }
    }

    /* cache miss: 遍历 type hierarchy */
    inst = object_dynamic_cast(obj, typename);

    if (!inst && obj) {
        fprintf(stderr, "...assertion failed...\n");
        abort();
    }

    if (!obj) {
        return inst;
    }

    /* 更新 cache */
    obj->class->object_cast_cache[i] = typename;
    return obj;

out:
    return obj;
}

object_dynamic_cast 的实质：

Object *object_dynamic_cast(Object *obj, const char *typename)
{
    TypeImpl *target_type = type_get_by_name_noload(typename);
    TypeImpl *obj_type = obj->class->type;

    if (type_is_ancestor(obj_type, target_type)) {
        return obj;
    }

    /* interface check: 遍历 obj_type 的所有接口，看是否匹配 target_type */
    for (int i = 0; i < obj_type->num_interfaces; i++) {
        ...
    }

    return NULL;
}

5.5 4 个 Cast 缓存槽

OBJECT_CLASS_CAST_CACHE = 4 意味着每个 Class 缓存最近 4 种成功的 cast。 这很重要：在热路径上（如每个 I/O 操作），避免每次都遍历 type hierarchy。 

struct ObjectClass {
    ...
    const char *object_cast_cache[4];  /* OBJECT_CHECK 缓存 */
    const char *class_cast_cache[4];   /* OBJECT_CLASS_CHECK 缓存 */
};

5.6 接口 (Interface)

QOM 支持类似 Java implements 的概念——接口：

/* 声明一个接口类型（通常是 abstract 的） */
#define TYPE_USER_CREATABLE "user-creatable"

typedef struct InterfaceClass UserCreatableClass {
    InterfaceClass parent_class;
    void (*complete)(UserCreatable *uc, Error **errp);
    bool (*can_be_deleted)(UserCreatable *uc);
} UserCreatableClass;

/* 在 OBJECT_DEFINE_TYPE_WITH_INTERFACES 中声明接口: */
OBJECT_DEFINE_TYPE_WITH_INTERFACES(MyDevice, my_device,
    MY_DEVICE, DEVICE,
    { TYPE_USER_CREATABLE },      /* interface 1 */
    { TYPE_RESETTABLE_INTERFACE }, /* interface 2 */
    { }                            /* sentinel */
);

类型检查时，接口也被纳入 type_is_ancestor 的检查范围：

/* 检查 obj 是否实现了 user-creatable 接口 */
UserCreatable *uc = (UserCreatable *)object_dynamic_cast(obj, TYPE_USER_CREATABLE);
if (uc) {
    user_creatable_complete(uc, errp);  /* 安全: 确认实现了该接口 */
}

5.7 OBJECT_DEFINE_TYPE 宏展开全景

这是理解 QOM 编程范式的 最核心的宏。以 TYPE_OBJECT → TYPE_DEVICE 为例： 

/* hw/core/qdev.c — 简化示意 */

/* OBJECT_DEFINE_TYPE(DeviceState, device, DEVICE, OBJECT) 展开为: */

static void device_finalize(Object *obj);
static void device_class_init(ObjectClass *oc, const void *data);
static void device_init(Object *obj);

static const TypeInfo device_info = {
    .parent          = TYPE_OBJECT,          /* ← 继承声明 */
    .name            = "device",
    .instance_size   = sizeof(DeviceState),
    .instance_align  = __alignof__(DeviceState),
    .instance_init   = device_init,
    .instance_finalize = device_finalize,
    .class_size      = sizeof(DeviceClass),
    .class_init      = device_class_init,
    .abstract        = true,               /* TYPE_DEVICE 是抽象的 */
};

static void device_register_types(void)
{
    type_register_static(&device_info);
}
type_init(device_register_types);         /* ★ 构造时注册 */

/* 然后开发者实现三个函数: */
static void device_init(Object *obj)       { /* 实例初始化 */ }
static void device_class_init(ObjectClass *oc, const void *data) {
    /* 设置虚函数指针, 注册属性 */
}
static void device_finalize(Object *obj)   { /* 清理 */ }
完整 QOM 编程范式的三个步骤：
  1. Header: 定义 Instance struct (首字段: ParentInstance)、Class struct (首字段: ParentClass)，调用 OBJECT_DECLARE_TYPE
  2. Source: 调用 OBJECT_DEFINE_TYPE 宏，实现 _init/_class_init/_finalize
  3. Registration: type_init 宏自动在 QOM 初始化阶段调用 type_register_static

本模块小结

机制实现
继承TypeInfo.parent → memcpy(parent_class) → class_init 覆盖

运行时类型检查type_is_ancestor(): 沿 parent 链向上查找，含 interface 检查
Cast 缓存object_cast_cache[4]: 缓存最近 4 次成功 cast 的结果
