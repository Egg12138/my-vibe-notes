#02 类型注册与发现

QOM 的核心是一个 全局类型注册表。每种 QOM 类型必须先注册到这个表中， 之后才能被查找和实例化。这一讲深入类型注册的完整链路。

2.1 TypeInfo —— 类型蓝图

TypeInfo 是向 QOM 注册类型的 "申请表"。它定义在 include/qom/object.h： 

/* include/qom/object.h:476 */
struct TypeInfo
{
    const char *name;             /* 类型字符串名称，必须是全局唯一的 */
    const char *parent;           /* 父类型的名称 */

    size_t instance_size;         /* 实例结构体的大小 (sizeof(MyDevice)) */
    size_t instance_align;        /* 实例内存对齐要求 */
    void (*instance_init)(Object *obj);
    void (*instance_post_init)(Object *obj);
    void (*instance_finalize)(Object *obj);

    bool abstract;                /* 抽象类型不能被直接实例化 */
    size_t class_size;            /* Class 结构体的大小 (sizeof(MyDeviceClass)) */

    void (*class_init)(ObjectClass *klass, const void *data);
    void (*class_base_init)(ObjectClass *klass, const void *data);
    const void *class_data;       /* 传递给 class_init/class_base_init 的额外数据 */

    const InterfaceInfo *interfaces;  /* 实现的接口列表 */
};

关键语义：

  • name 是类型的 全限定名，类似于 C++ 的 typeid(MyClass).name()
  • parent 指向父类型的 name —— QOM 的继承链就是通过 字符串 串联的
  • class_size = 0 表示该类型不增加任何 Class 字段，复用父类的 Class 大小
  • instance_size = 0 → 自动标记为 abstract = true
  • class_init 在该类型 自身 的 class 准备好后被调用（覆盖虚函数）
  • class_base_init 在该类型 被用作父类型时 被调用（撤销 memcpy 的影响）
class_base_init 的微妙用途：当子类型通过 memcpy(child_class, parent_class, parent_class_size) 继承了父类的 class 内容后， 父类的 class_base_init 被调用，给父类一个机会 撤销 某些不应当被继承的字段。 例如一个 class 成员 bool subclassed，父类自身应设为 false，但当子类继承后应设为 true。

2.2 TypeImpl —— 内部类型表示

TypeInfo 是公开的 "申请表"，QOM 内部用 TypeImpl 存储已注册的类型。 定义在 qom/object.c： 

/* qom/object.c:48 */
struct TypeImpl
{
    const char *name;

    size_t class_size;
    size_t instance_size;
    size_t instance_align;

    void (*class_init)(ObjectClass *klass, const void *data);
    void (*class_base_init)(ObjectClass *klass, const void *data);
    const void *class_data;

    void (*instance_init)(Object *obj);
    void (*instance_post_init)(Object *obj);
    void (*instance_finalize)(Object *obj);

    bool abstract;

    const char *parent;
    TypeImpl *parent_type;  /* 懒加载：首次 type_get_parent() 时解析 */

    ObjectClass *class;        /* 懒初始化：首次 type_initialize() 时分配 */

    int num_interfaces;
    InterfaceImpl interfaces[MAX_INTERFACES];  /* 最多 32 个接口 */
};

TypeImpl 与 TypeInfo 几乎是一一对应的，关键差异在于 TypeImpl 包含 运行时解析结果：

  • parent_type —— 从 parent 字符串解析出的实际 TypeImpl 指针
  • class —— 分配好的 ObjectClass，在首次 type_initialize 时创建
  • instance_size 和 class_size —— 经过继承链计算后的最终大小

2.3 类型注册流程

type_register_static(&my_type_info) │ ├─ assert(info->parent != NULL) │ └─ type_register_internal(info) ├─ type_name_is_valid(name) /* 类型名必须是合法的 C identifier */ ├─ ti = type_new(info) /* 从 TypeInfo 复制字段到新 TypeImpl */ └─ type_table_add(ti) /* 插入全局 GHashTable */

核心实现在 qom/object.c:163：

static TypeImpl *type_register_internal(const TypeInfo *info)
{
    TypeImpl *ti;

    if (!type_name_is_valid(info->name)) {
        fprintf(stderr, "Registering '%s' with illegal type name\n",
                info->name);
        abort();
    }

    ti = type_new(info);
    type_table_add(ti);
    return ti;
}

TypeImpl *type_register_static(const TypeInfo *info)
{
    assert(info->parent);   /* 所有非根类型必须有 parent */
    return type_register_internal(info);
}

type_new() 负责将 TypeInfo 各字段复制到新分配的 TypeImpl，但 不解析 parent —— 这被推迟到首次 type_get_parent() 时。 type_table_add() 将 TypeImpl 插入全局哈希表，键为 name 字符串。

2.4 type_init —— 连接编译与注册

每个类型的注册函数如何被调用？答案是 type_init 宏和 GCC 的 __attribute__((constructor))： 

/* include/qemu/module.h:56 */
#define type_init(function) module_init(function, MODULE_INIT_QOM)

/* module_init 展开后 (简化版 on Linux): */
#define module_init(function, type)                              \
    static void __attribute__((constructor)) do_qemu_init_ ## function(void) \
    {                                                            \
        register_module_init(function, type);                    \
    }
main() │ ├─ (constructor functions run before main) │ └─ register_module_init(my_register_types, MODULE_INIT_QOM) │ └─ 将 my_register_types 加入 init_funcs[MODULE_INIT_QOM] 链表 │ └─ qemu_init() └─ module_call_init(MODULE_INIT_QOM) └─ 遍历 init_funcs[MODULE_INIT_QOM] 链表 └─ 调用 my_register_types() └─ type_register_static(&my_type_info) ★ 真正注册发生在这里！

关键点：type_init 不直接注册类型 —— 它只是把注册函数挂到一个队列里， 等到 module_call_init(MODULE_INIT_QOM) 时统一执行。 这样能保证所有 module 的 constructor 跑完后，才统一注册和初始化类型。

启动顺序保证：QEMU 按特定顺序调用 module_call_init()： MIGRATION → BLOCK → OPTS → QOM → TRACE → XEN_BACKEND → LIBQOS → FUZZ_TARGET。 QOM 在 OPTS 之后、TRACE 之前，所以 QOM 类型可以引用 opts 解析出的配置。

2.5 类型的查找与动态加载

/* 简单查找 —— 只在已注册的表中查，不触发 module 加载 */
static TypeImpl *type_get_by_name_noload(const char *name)
{
    if (name == NULL) return NULL;
    return type_table_lookup(name);
}

/* 完整查找 —— 找不到时尝试加载 module */
static TypeImpl *type_get_or_load_by_name(const char *name, Error **errp)
{
    TypeImpl *type = type_get_by_name_noload(name);
#ifdef CONFIG_MODULES
    if (!type) {
        int rv = module_load_qom(name, errp);  /* 尝试加载 .so 文件 */
        if (rv > 0) {
            type = type_get_by_name_noload(name);   /* 模块加载后重试 */
        }
    }
#endif
    if (!type) {
        error_setg(errp, "unknown type '%s'", name);
    }
    return type;
}

module_load_qom() 是 QEMU 的模块系统：当类型不在已注册表中时， 它会尝试加载 hw-arm-virt.so 或 hw-usb-host.so 这样的共享库。 这些 .so 文件在加载时执行其 constructor（即 type_init 注册的函数）， 从而向全局表注册新的类型。

2.6 类型名的命名约定

QOM 类型名遵循严格的模板风格：

模式示例说明
"object"TYPE_OBJECT根类型
"device"TYPE_DEVICE设备基类
"bus"TYPE_BUS总线基类
"pci-device"TYPE_PCI_DEVICEPCI 设备
"sys-bus-device"TYPE_SYS_BUS_DEVICE系统总线设备
"machine"TYPE_MACHINEMachine 类型
"memory-backend-ram"TYPE_MEMORY_BACKEND_RAM内存后端
"<vendor>-<device>""e1000", "virtio-net-pci"具体设备

对应的 C 宏约定：

  • TYPE_MODULE_OBJ_NAME → 类型名字符串 (如 TYPE_DEVICE → "device")
  • OBJECT_DECLARE_TYPE(DeviceState, DeviceClass, DEVICE) → 生成 cast helpers
  • OBJECT_DEFINE_TYPE(DeviceState, device, DEVICE, OBJECT) → 生成 TypeInfo + 注册函数

本模块小结

概念含义
TypeInfo公开的注册 "申请表"，描述类型的元信息
TypeImpl内部的类型存储，包含懒解析的 parent_type 和 class
